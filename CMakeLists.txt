cmake_minimum_required(VERSION 3.20)
project(FolderIcon LANGUAGES C CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "FolderIcon requires Windows")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Keep the upstream C implementation buildable as a separate target.
add_executable(FolderIconC WIN32 main.c)
target_compile_definitions(FolderIconC PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
target_link_libraries(FolderIconC PRIVATE user32 shell32 gdi32 comctl32 dwmapi uxtheme ole32)

add_executable(FolderIconCN WIN32 src/entry_actions.cpp src/entry_maintenance.cpp
    src/main.cpp src/shell.cpp src/config_store.cpp src/context_menu.cpp src/panel.cpp src/settings.cpp src/icon_cache.cpp src/folder_sort.cpp src/folder_rules.cpp src/sort_settings.cpp src/folder_badge.cpp src/app.rc)
target_compile_definitions(FolderIconCN PRIVATE _CRT_SECURE_NO_WARNINGS)
target_link_libraries(FolderIconCN PRIVATE user32 shell32 gdi32 comctl32 comdlg32 ole32 oleaut32 oleacc uuid advapi32 dwmapi shlwapi gdiplus bcrypt)
if(MSVC)
    target_compile_options(FolderIconCN PRIVATE /utf-8)
elseif(MINGW)
    target_link_options(FolderIconCN PRIVATE -municode)
endif()

include(CTest)
